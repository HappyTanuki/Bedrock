function(_BedrockBuildConfig_register_targets)
    cmake_parse_arguments(ARG "" "INSTALL_DESTINATION" "" ${ARGN})
    if(NOT ARG_INSTALL_DESTINATION)
        message(FATAL_ERROR "_BedrockBuildConfig_register_targets requires INSTALL_DESTINATION")
    endif()
    get_property(_registered GLOBAL PROPERTY _BEDROCK_BUILD_CONFIG_REGISTERED)
    if(_registered)
        return()
    endif()
    _BedrockBuildConfig_get_targets(_targets)
    install(TARGETS ${_targets} EXPORT _BedrockBuildConfigTargets)
    install(EXPORT _BedrockBuildConfigTargets NAMESPACE _BedrockBuildConfig:: DESTINATION "${ARG_INSTALL_DESTINATION}")
    configure_package_config_file("${CMAKE_CURRENT_FUNCTION_LIST_DIR}/_BedrockBuildConfigConfig.cmake.in" "${CMAKE_CURRENT_BINARY_DIR}/_BedrockBuildConfigConfig.cmake" INSTALL_DESTINATION "${ARG_INSTALL_DESTINATION}")
    install(FILES "${CMAKE_CURRENT_BINARY_DIR}/_BedrockBuildConfigConfig.cmake" DESTINATION "${ARG_INSTALL_DESTINATION}")
    set_property(GLOBAL PROPERTY _BEDROCK_BUILD_CONFIG_REGISTERED TRUE)
endfunction()
function(_BedrockBuildConfig_export_build_tree)
    cmake_parse_arguments(ARG "" "DESTINATION" "" ${ARGN})
    if(NOT ARG_DESTINATION)
        message(FATAL_ERROR "_BedrockBuildConfig_export_build_tree requires DESTINATION")
    endif()
    _BedrockBuildConfig_get_targets(_targets)
    export(TARGETS ${_targets} NAMESPACE _BedrockBuildConfig:: FILE "${ARG_DESTINATION}")
endfunction()
